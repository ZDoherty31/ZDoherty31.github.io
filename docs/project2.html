-- Calculate profit as revenue minus budget
-- Filter out rows where budget or revenue is missing (0)
SELECT title,
       year,
       budget,
       revenue,
       revenue - budget AS profit
FROM tmdb
WHERE revenue > 0 AND budget > 0
ORDER BY profit DESC

Query 2 - Average IMDb Rating and Metascore by Genre

Done by Zoe

-- Join movies and movie_ratings on title and year to match films correctly
-- Group by genre to get average ratings per genre
SELECT m.genre,
       ROUND(AVG(r.imdb), 2) AS avg_imdb,
       ROUND(AVG(r.metascore), 2) AS avg_metascore,
       COUNT(*) AS movie_count
FROM movie_ratings r
JOIN movies m ON r.title = m.title
             AND r.year = m.year
GROUP BY m.genre
ORDER BY avg_imdb DESC

Done by Zoe

-- Group movies into decades using integer division (e.g., 1987 / 10 * 10 = 1980)
-- Filter out rows with missing revenue
SELECT (year / 10) * 10 AS decade,
       ROUND(AVG(popularity), 2) AS avg_popularity,
       ROUND(AVG(revenue), 2) AS avg_revenue,
       COUNT(*) AS movie_count
FROM tmdb
WHERE year IS NOT NULL
  AND revenue > 0
GROUP BY decade
ORDER BY decade

-- Join all three tables to combine genre/director info with ratings and TMDB metrics

Owen’s Queries - Movie Ratings and Genre Trends Over Time

-- Join tmdb and movies to combine release year, content rating, genre, and vote average
-- Filter to only the four major content ratings for clarity
CREATE OR REPLACE TABLE table1 AS
SELECT EXTRACT(YEAR FROM CAST(release_date AS DATE)) AS year,
       FLOOR(EXTRACT(YEAR FROM CAST(release_date AS DATE)) / 10) * 10 AS decade,
       tmdb.title,
       movies.rating,
       movies.genre,
       vote_average
FROM tmdb
JOIN movies ON tmdb.title = movies.title
WHERE movies.rating IN ('R', 'PG', 'PG-13', 'G')
ORDER BY year
-- Count movies per year per rating and compute average votes
-- Used ChatGPT assistance to figure out the CROSS JOIN and proportion logic
CREATE OR REPLACE TABLE table2 AS
WITH base AS (
    SELECT year, rating,
           COUNT(*) AS count,
           AVG(vote_average) AS avg_votes
    FROM table1
    GROUP BY year, rating
),
years AS (SELECT DISTINCT year FROM table1),
ratings AS (SELECT DISTINCT rating FROM table1),
all_combos AS (
    SELECT y.year, r.rating
    FROM years y CROSS JOIN ratings r
)
SELECT a.year, a.rating,
       COALESCE(b.count, 0) AS count,
       COALESCE(b.count, 0) * 1.0
           / SUM(COALESCE(b.count, 0)) OVER (PARTITION BY a.year) AS proportion,
       b.avg_votes
FROM all_combos a
LEFT JOIN base b ON a.year = b.year AND a.rating = b.rating
WHERE count != 0
ORDER BY a.year
-- Same structure as table2 but grouped by genre instead of rating
CREATE OR REPLACE TABLE table3 AS
WITH base AS (
    SELECT year, genre,
           COUNT(*) AS count,
           AVG(vote_average) AS avg_votes
    FROM table1
    GROUP BY year, genre
),
years AS (SELECT DISTINCT year FROM table1),
genres AS (SELECT DISTINCT genre FROM table1),
all_combos AS (
    SELECT y.year, g.genre
    FROM years y CROSS JOIN genres g
)
SELECT a.year, a.genre,
       COALESCE(b.count, 0) AS count,
       COALESCE(b.count, 0) * 1.0
           / SUM(COALESCE(b.count, 0)) OVER (PARTITION BY a.year) AS proportion,
       b.avg_votes
FROM all_combos a
LEFT JOIN base b ON a.year = b.year AND a.genre = b.genre
WHERE count != 0
ORDER BY a.year
-- Combine table2 and table3 into a single summary table for visualization
CREATE OR REPLACE TABLE table4 AS
SELECT table2.year,
       table2.rating,
       table2.proportion AS rating_prop,
       table3.genre,
       table3.proportion AS genre_prop,
       AVG(table2.avg_votes) AS avg_votes
FROM table2
JOIN table3 ON table2.year = table3.year
GROUP BY table2.year, table2.rating, rating_prop, table3.genre, genre_prop

Visualization 1 - Average IMDb Rating by Genre

# Pull the genre ratings summary from the database into R for plotting
genre_ratings <- dbGetQuery(con, "
  SELECT m.genre,
         ROUND(AVG(r.imdb), 2) AS avg_imdb,
         COUNT(*) AS movie_count
  FROM movie_ratings r
  JOIN movies m ON r.title = m.title
               AND r.year = m.year
  GROUP BY m.genre
  ORDER BY avg_imdb DESC
")
# Bar chart of average IMDb rating by genre
# reorder() sorts genres from lowest to highest avg_imdb for readability
# coord_flip() makes genre labels horizontal and easier to read
# scale_fill_gradient applies a blue color gradient - darker = higher rating
ggplot(genre_ratings, aes(x = reorder(genre, avg_imdb), y = avg_imdb, fill = avg_imdb)) +
  geom_col() +
  coord_flip() +
  scale_fill_gradient(low = "lightblue", high = "darkblue") +
  labs(
    title = "Average IMDb Rating by Genre",
    subtitle = "Based on movies appearing in both datasets",
    x = "Genre",
    y = "Average IMDb Rating",
    fill = "Avg Rating"
  ) +
  theme_minimal()
